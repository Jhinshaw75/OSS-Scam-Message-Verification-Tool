// Enhanced Scam Keyword Triggers (including aggressive legal/asset threats and punctuation stripping) const scamKeywords = [ "final attempt", "resolve your current matter", "filing the claim", "order against your assets", "contact our office immediately", "leon county", "legal action" ]; // Function to check message text against triggers with hyphens/punctuation stripped function analyzeMessageText(text) { // Strip hyphens, periods, and extra punctuation between letters to prevent evasion tricks const cleanedText = text.toLowerCase().replace(/[-.,/#!$%^&*;:{}=\-_`~()]/g, ''); // Using .filter array method to scan for red flags on the cleaned string const matchedThreats = scamKeywords.filter(keyword => cleanedText.includes(keyword.replace(/[-.,/#!$%^&*;:{}=\-_`~()]/g, ''))); if (matchedThreats.length > 0) { return { isScam: true, riskLevel: "High", matched: matchedThreats, message: "Warning: High likelihood of a legal scare scam or predatory debt collection tactic." }; } return { isScam: false, riskLevel: "Low", message: "No immediate red flags detected, but always verify independently." }; }
