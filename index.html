function analyzeMessage() { const rawText = document.getElementById("scamInput").value.trim(); const resultContainer = document.getElementById("resultContainer"); if (!rawText) { alert("Please paste text or upload a screenshot/document to analyze."); return; } const normalizedText = rawText.replace(/-/g, '').toLowerCase(); // Use whole-word matching (splitting text into words) to prevent false positives like "irs" inside "department" const words = normalizedText.match(/\b[a-z0-9]+\b/g) || []; const scamKeywords = [ "urgent", "locked", "bank", "verify", "suspend", "winner", "prize", "gift card", "social security", "irs", "fbi", "click here", "password", "crypto", "wire", "attachment", "screenshot", "warrant", "federal", "arrest", "compliance", "directive", "termination", "audit", "jurisdiction", "penalty", "backflow", "assembly", "lapsed" ]; let isScam = false; let matchedKeyword = ""; // Check for outreach/social engineering pretexts or whole-word keyword matches const hasUrl = normalizedText.includes("http") || normalizedText.includes("www.") || normalizedText.includes(".net") || normalizedText.includes(".com"); const hasSocialEngineering = (normalizedText.includes("outreach") || normalizedText.includes("grant") || normalizedText.includes("wellness") || normalizedText.includes("community")) && (hasUrl || normalizedText.includes("confirm") || normalizedText.includes("schedule")); if (hasSocialEngineering) { isScam = true; matchedKeyword = "social engineering / community outreach pretext"; } else { for (let kw of scamKeywords) { // Check if the keyword exists as a whole word in the text array const kwParts = kw.split(" "); if (kwParts.length === 1) { if (words.includes(kw)) { isScam = true; matchedKeyword = kw; break; } } else { // For multi-word phrases like "social security" if (normalizedText.includes(kw)) { isScam = true; matchedKeyword = kw; break; } } } } let outputHTML = ""; if (isScam || rawText.length < 15) { outputHTML = `

⚠️ STOP. POTENTIAL SCAM DETECTED.

Threat Vector Identified: Suspicious pattern matched ("${matchedKeyword}"). This message or attached file exhibits classic signs of digital fraud, social engineering pretexts, or unauthorized credential harvesting.

Required Safety Actions:

  1. Do Not Click Any Links: Tapping links in this message or attachment could compromise your device.
  2. Do Not Call Provided Numbers: Do not dial phone numbers listed inside this notice. Scammers staff these lines directly.
  3. Verify Independently: If this claims to be from your bank, utility, or government agency, look at the back of your actual card or official statement and call that verified number directly.
  4. Delete the Message: Remove the threat from your inbox or phone immediately.

Official Florida & Federal Help Resources:

  • 📞 DOEA Elder Helpline: 1-800-96-ELDER (1-800-963-5337)
  • 📞 Florida Abuse Hotline (Financial Exploitation): 1-800-96-ABUSE
  • 🌐 Operation Senior Shield: opseniorshieldfl.com
  • ⚖️ FTC Reporting Portal: reportfraud.ftc.gov
`; } else { outputHTML = `

✅ NO IMMEDIATE THREAT DETECTED

The analyzed text or attachment does not contain common high-risk fraud keywords or emergency pressure tactics. However, always exercise normal digital caution before sharing personal or financial information.

Need Guidance?

If you still feel uneasy about this message, you can always contact the Florida Department of Elder Affairs Helpline at 1-800-96-ELDER.

`; } resultContainer.style.display = "block"; resultContainer.innerHTML = outputHTML; resultContainer.scrollIntoView({ behavior: 'smooth' }); }
